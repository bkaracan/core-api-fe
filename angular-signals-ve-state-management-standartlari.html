Zoneless Değişiklik Algılama Standartları

Zoneless Zorunluluğu

src/app/app.config.ts (Zoneless Provider)
TypeScript
import { ApplicationConfig, provideExperimentalZonelessChangeDetection } from '@angular/core';

export const appConfig: ApplicationConfig = {
  providers: [
    provideExperimentalZonelessChangeDetection()
  ]
};

Signal, Computed ve Effect Disiplini

Durum mutasyonu ve türetilmiş durum hesaplamaları katı kurallara bağlanmıştır.

src/app/features/users/user-profile.component.ts
TypeScript
import { Component, signal, computed, effect, ChangeDetectionStrategy } from '@angular/core';

@Component({
  selector: 'app-user-profile',
  standalone: true,
  changeDetection: ChangeDetectionStrategy.OnPush,
  template: `
    <div>
      <h3>{{ fullName() }}</h3>
      <p>Oturum Durumu: {{ isAdmin() ? 'Yönetici' : 'Standart Kullanıcı' }}</p>
      <button (click)="incrementLogin()">Giriş Sayısını Artır: {{ loginCount() }}</button>
    </div>
  `
})
export class UserProfileComponent {
  // 1. Yazılabilir (Writable) Sinyaller
  readonly firstName = signal<string>('Ahmet');
  readonly lastName = signal<string>('Yılmaz');
  readonly roles = signal<string[]>(['ROLE_ADMIN']);
  readonly loginCount = signal<number>(1);

  // 2. Türetilmiş (Computed) Sinyaller - Memoized ve saf (pure)
  readonly fullName = computed(() => `${this.firstName()} ${this.lastName()}`);
  readonly isAdmin = computed(() => this.roles().includes('ROLE_ADMIN'));

  constructor() {
    // 3. Saf Yan Etki (Effect) - Kesinlikle durum mutasyonu içermez
    effect(() => {
      console.log(`Kullanıcı ${this.fullName()} oturum sayısı: ${this.loginCount()}`);
    });
  }

  incrementLogin(): void {
    this.loginCount.update(c => c + 1);
  }
}

Yeni Sinyal Tabanlı Component I/O Standartları

Eski @Input() ve @Output() dekoratörleri yasaktır. Tip-güvenli input(), output() ve model() kullanılmalıdır.

src/app/shared/components/search-box/search-box.component.ts
TypeScript
import { Component, input, output, model } from '@angular/core';

@Component({
  selector: 'app-search-box',
  standalone: true,
  template: `
    <input
      [value]="searchTerm()"
      (input)="onInputChange($event)"
      [placeholder]="placeholder()" />
    <button (click)="clear.emit()">Temizle</button>
  `
})
export class SearchBoxComponent {
  // Zorunlu olmayan girdi
  readonly placeholder = input<string>('Arama yapın...');

  // Çift yönlü iki-yönlü sinyal modeli [(searchTerm)]
  readonly searchTerm = model<string>('');

  // Olay çıktısı
  readonly clear = output<void>();

  onInputChange(event: Event): void {
    const val = (event.target as HTMLInputElement).value;
    this.searchTerm.set(val);
  }
}

Resource API & rxResource ile Asenkron Veri Yükleme

Angular 21 rxResource, filtre ve sayfa sinyalleri değiştikçe eski istekleri iptal ederek otomatik veri çeker.

src/app/core/services/user-data.service.ts
TypeScript
import { Injectable, signal, computed, inject } from '@angular/core';
import { HttpClient, HttpParams } from '@angular/common/http';
import { rxResource } from '@angular/core/rxjs-interop';

@Injectable({ providedIn: 'root' })
export class UserDataService {
  private readonly http = inject(HttpClient);

  readonly query = signal<string>('');
  readonly page = signal<number>(0);

  readonly userResource = rxResource({
    request: () => ({ q: this.query(), p: this.page() }),
    loader: ({ request }) => {
      const params = new HttpParams().set('query', request.q).set('page', request.p);
      return this.http.get<any>('/api/v1/users', { params });
    }
  });

  readonly users = computed(() => this.userResource.value()?.data ?? []);
  readonly isLoading = this.userResource.isLoading;
}

Kurumsal Custom Signal Store Mimarisi

Karmaşık NgRx boilerplate'i yerine saf TypeScript ve Signal primitifleri ile kurumsal hafif store oluşturulur.

src/app/core/state/user.store.ts
TypeScript
import { Injectable, signal, computed } from '@angular/core';

export interface UserState {
  users: any[];
  selectedUserId: string | null;
  filter: string;
  loading: boolean;
}

const initialState: UserState = {
  users: [],
  selectedUserId: null,
  filter: '',
  loading: false
};

@Injectable({ providedIn: 'root' })
export class UserStore {
  // Gizli State
  private readonly _state = signal<UserState>(initialState);

  // Public Seçiciler (Selectors)
  readonly users = computed(() => this._state().users);
  readonly selectedUser = computed(() =>
    this._state().users.find(u => u.id === this._state().selectedUserId) ?? null
  );
  readonly loading = computed(() => this._state().loading);

  // Eylemler (Mutations / Actions)
  setUsers(users: any[]): void {
    this._state.update(s => ({ ...s, users, loading: false }));
  }

  selectUser(id: string): void {
    this._state.update(s => ({ ...s, selectedUserId: id }));
  }

  setLoading(loading: boolean): void {
    this._state.update(s => ({ ...s, loading }));
  }
}

RxJS & Signal Birlikte Yaşama Standartları

Arama debounce gibi akışlarda RxJS kullanılır; bileşene taşınırken toSignal ve takeUntilDestroyed standarttır.

src/app/features/users/user-search.component.ts
TypeScript
import { Component, inject } from '@angular/core';
import { FormControl, ReactiveFormsModule } from '@angular/forms';
import { toSignal, takeUntilDestroyed } from '@angular/core/rxjs-interop';
import { debounceTime, distinctUntilChanged, switchMap } from 'rxjs';
import { HttpClient } from '@angular/common/http';

@Component({
  selector: 'app-user-search',
  standalone: true,
  imports: [ReactiveFormsModule],
  template: `
    <input [formControl]="searchControl" placeholder="Arayın..." />
    <ul>
      @for (item of searchResults(); track item.id) {
        <li>{{ item.name }}</li>
      }
    </ul>
  `
})
export class UserSearchComponent {
  private readonly http = inject(HttpClient);
  readonly searchControl = new FormControl('', { nonNullable: true });

  // RxJS akışını otomatik temizlenen (unsubscribed) Signal'e dönüştür
  readonly searchResults = toSignal(
    this.searchControl.valueChanges.pipe(
      debounceTime(300),
      distinctUntilChanged(),
      switchMap(term => this.http.get<any[]>(`/api/v1/users/search?q=${term}`)),
      takeUntilDestroyed()
    ),
    { initialValue: [] }
  );
}
